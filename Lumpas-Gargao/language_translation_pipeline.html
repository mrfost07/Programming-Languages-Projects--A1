# stack.py — Stack data structure used as the token buffer in the pipeline

class Stack:
    """
    LIFO Stack — Last-In, First-Out token memory buffer.
    Stores tokenized words before processing by translation agents.
    """

    def __init__(self):
        self._data: list[str] = []

    def push(self, token: str) -> None:
        """Push a token onto the top of the stack."""
        self._data.append(token)

    def pop(self) -> str:
        """Remove and return the top token (LIFO order)."""
        if self.is_empty():
            raise IndexError("pop from empty Stack")
        return self._data.pop()

    def peek(self) -> str | None:
        """Return the top token without removing it."""
        return self._data[-1] if self._data else None

    def is_empty(self) -> bool:
        return len(self._data) == 0

    def depth(self) -> int:
        return len(self._data)

    def drain(self) -> list[str]:
        """Pop all tokens from stack in LIFO order."""
        result = []
        while not self.is_empty():
            result.append(self.pop())
        return result

    def __repr__(self) -> str:
        return f"Stack(depth={self.depth()}, top={self.peek()!r})"


def tokenize(text: str) -> list[str]:
    """
    Tokenize a sentence into clean word tokens.
    Strips punctuation and splits on whitespace.
    """
    import re
    cleaned = re.sub(r"[^\w\s]", "", text)
    return [tok for tok in cleaned.split() if tok]

# ── Analysis Result Dataclass ─────────────────────────────────────────

@dataclass
class Analysis:
    tokens: list[str]
    sentence_type: str   # "interrogative" | "declarative" | "exclamatory"
    sentiment: str        # "positive" | "neutral" | "negative"
    word_count: int
    intent: str
    raw_text: str


# ── CLAUDE — Semantic Analysis Agent ──────────────────────────────────

class Claude:
    """
    Claude reads the full token stream from the Stack and performs
    semantic analysis: detecting sentence type, sentiment, and intent.
    This metadata helps Jules choose the correct translation register.
    """

    POSITIVE_WORDS = {"love", "great", "good", "fun", "happy", "wonderful"}
    NEGATIVE_WORDS = {"bad", "hate", "terrible", "awful", "horrible"}

    def analyze(self, tokens: list[str], raw_text: str) -> Analysis:
        """Perform semantic analysis on the token stream."""
        lower = {t.lower() for t in tokens}

        # Determine sentence type
        if raw_text.rstrip().endswith("?"):
            stype = "interrogative"
        elif raw_text.rstrip().endswith("!"):
            stype = "exclamatory"
        else:
            stype = "declarative"

        # Detect sentiment
        if lower & self.POSITIVE_WORDS:
            sentiment = "positive"
        elif lower & self.NEGATIVE_WORDS:
            sentiment = "negative"
        else:
            sentiment = "neutral"

        # Infer intent
        greetings = {"hello", "hi", "hey", "good", "morning"}
        if lower & greetings:
            intent = "greeting"
        elif stype == "interrogative":
            intent = "question"
        else:
            intent = "statement"

        return Analysis(
            tokens=tokens,
            sentence_type=stype,
            sentiment=sentiment,
            word_count=len(tokens),
            intent=intent,
            raw_text=raw_text,
        )


# ── JULES — Lexical Translation Engine ────────────────────────────────

class Jules:
    """
    Jules pops each token from the Stack (LIFO), looks it up in the
    language lexicon, and produces the translated token list.
    Tokens not found in the lexicon are wrapped in brackets.
    """

    def translate(self, stack: Stack, target_lang: str) -> list[str]:
        """Drain the stack and translate each token."""
        dictionary = LEXICON.get(target_lang, {})
        popped_tokens: list[str] = stack.drain()  # LIFO order
        translated: list[str] = []

        for token in reversed(popped_tokens):  # restore original order
            key = token[0].upper() + token[1:]  # try capitalized first
            result = (
                dictionary.get(token)
                or dictionary.get(key)
                or dictionary.get(token.lower())
                or f"[{token}]"
            )
            translated.append(result)

        return translated

    def coverage(self, tokens: list[str], translated: list[str]) -> float:
        """Return what fraction of tokens were found in the lexicon."""
        known = sum(1 for t in translated if not t.startswith("["))
        return known / len(tokens) if tokens else 0.0


# ── STITCH — Grammar & Assembly Agent ─────────────────────────────────

class Stitch:
    """
    Stitch receives the translated token list and an Analysis object,
    then reassembles the tokens into a grammatically correct sentence.
    It restores punctuation and capitalizes the output.
    """

    def assemble(self, translated: list[str], analysis: Analysis) -> str:
        """Join translated tokens and apply grammar rules."""
        if not translated:
            return ""

        sentence = " ".join(translated)

        # Restore terminal punctuation
        punctuation_map = {
            "interrogative": "?",
            "exclamatory":   "!",
            "declarative":   ".",
        }
        sentence += punctuation_map.get(analysis.sentence_type, ".")

        # Capitalize first character
        sentence = sentence[0].upper() + sentence[1:]

        return sentence

# pipeline.py — Orchestrates the full translation pipeline

from dataclasses import dataclass
from stack   import Stack, tokenize
from agents  import Claude, Jules, Stitch, Analysis


@dataclass
class PipelineResult:
    source_text:   str
    target_lang:   str
    tokens:        list[str]
    analysis:      Analysis
    translated:    list[str]
    output:        str
    coverage:      float


class TranslationPipeline:
    """
    Orchestrates all four components of the translation process:

        ┌────────┐    push()    ┌────────┐
        │ Input  │ ──────────► │ Stack  │
        └────────┘             └────┬───┘
                                    │  stream tokens
                               ┌────▼────┐
                               │ Claude  │  semantic analysis
                               └────┬────┘
                                    │  Analysis object
                               ┌────▼────┐
                               │  Jules  │  lexical translation
                               └────┬────┘
                                    │  translated tokens
                               ┌────▼────┐
                               │ Stitch  │  grammar & assembly
                               └────┬────┘
                                    │
                               ┌────▼────┐
                               │ Output  │
                               └─────────┘
    """

    def __init__(self):
        self.stack  = Stack()
        self.claude = Claude()
        self.jules  = Jules()
        self.stitch = Stitch()

    def run(self, source_text: str, target_lang: str) -> PipelineResult:
        """Execute the full translation pipeline."""

        # ── Stage 1: Tokenize ──────────────────────────────────────────
        print(f"\n[Stack]  Tokenizing: {source_text!r}")
        tokens = tokenize(source_text)

        # ── Stage 2: Push to Stack ─────────────────────────────────────
        for token in tokens:
            self.stack.push(token)
            print(f"[Stack]  push({token!r}) → depth={self.stack.depth()}")

        # ── Stage 3: Claude analyzes ───────────────────────────────────
        analysis = self.claude.analyze(tokens, source_text)
        print(f"\n[Claude] type={analysis.sentence_type}, "
              f"sentiment={analysis.sentiment}, intent={analysis.intent}")

        # ── Stage 4: Jules translates ──────────────────────────────────
        translated = self.jules.translate(self.stack, target_lang)
        coverage   = self.jules.coverage(tokens, translated)
        print(f"\n[Jules]  {tokens} → {translated}")
        print(f"[Jules]  Coverage: {coverage:.0%}")

        # ── Stage 5: Stitch assembles ──────────────────────────────────
        output = self.stitch.assemble(translated, analysis)
        print(f"\n[Stitch] Output: {output!r}")

        return PipelineResult(
            source_text=source_text,
            target_lang=target_lang,
            tokens=tokens,
            analysis=analysis,
            translated=translated,
            output=output,
            coverage=coverage,
        )

# main.py — Entry point: run the Language Translation Pipeline

from pipeline import TranslationPipeline


def main():
    pipeline = TranslationPipeline()

    # ── Test phrases ───────────────────────────────────────────────────
    test_cases = [
        ("Hello, how are you doing today?", "Spanish"),
        ("Good morning world!",              "French"),
        ("I love programming language",      "Japanese"),
        ("What is your name?",               "German"),
    ]

    print("=" * 60)
    print("  Language Translation Pipeline")
    print("  Stack · Claude · Jules · Stitch")
    print("=" * 60)

    for source, lang in test_cases:
        print(f"\n{'─'*60}")
        print(f"Source ({lang}): {source}")
        print("─" * 60)

        result = pipeline.run(source, lang)

        print(f"\n✓ Result : {result.output}")
        print(f"  Tokens : {result.tokens}")
        print(f"  Intent : {result.analysis.intent}")
        print(f"  Coverage: {result.coverage:.0%}")


if __name__ == "__main__":
    main()
