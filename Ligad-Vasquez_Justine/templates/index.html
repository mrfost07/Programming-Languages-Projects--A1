Accepted: Duck typing
class Duck:
    def quack(self):
        return 'Quack!'

class Person:
    def quack(self):
        return 'I can imitate a duck!' 

obj = Person()
print(obj.quack())
Rejected: Wrong method
class Dog:
    def bark(self):
        return 'Woof!'

obj = Dog()
print(obj.quack())
